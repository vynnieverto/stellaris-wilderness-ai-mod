alloys_expenditure_ships = {
	resource = alloys
	type = expenditure
	category = ships
	potential = {
		always = yes
	}
	weight = {
		weight = 1
		modifier = {
			factor = 4
			stellarai_should_rebuild_fleet = yes
		}


		# spend less alloys on ships if over capacity
		modifier = {
			factor = 0.45
			is_at_war = no
			stellarai_machine_energy_emergency = yes
		}
		modifier = {
			factor = 0.85
			is_at_war = no
			stellarai_machine_minerals_emergency = yes
		}
		modifier = {
			factor = 1.15
			stellarai_machine_basic_econ_stable = yes
			used_naval_capacity_percent < 1
		}
		modifier = {
			factor = 0.6
			is_at_war = no
			stellarai_economic_crisis = yes
		}
		modifier = {
			factor = 0.65
			AND = {
				used_naval_capacity_percent >= 1
				is_at_war = no
			}
		}
		modifier = {
			factor = 0.82
			AND = {
				used_naval_capacity_percent >= 1.15
				is_at_war = yes
			}
		}
		# Biological Ships use less alloys
		modifier = {
			factor = 0.33
			country_uses_bio_ships = yes
		}
		modifier = {
			factor = 5
			used_naval_capacity_percent < 1.5
			OR = {
				any_neighbor_country = {
					has_ascension_perk = ap_become_the_crisis
				}
				any_country = {
					is_crisis_faction = yes
				}
			}
		}
	}
	desired_min = {
		base = 0
		modifier = {
			add = @ai_ship_budget_desired_min_per_year
			country_uses_bio_ships = no
		}
		modifier = {
			factor = trigger:years_passed
		}
	}
}

alloys_expenditure_ship_upgrades = {
	resource = alloys
	type = expenditure
	category = ship_upgrades
	potential = {
		is_at_war = no
		any_owned_fleet = {
			controller = {
				is_same_value = root
			}
			can_be_upgraded = yes
		}
	}
	weight = {
		weight = 0.2
		modifier = {
			factor = 0.25
			stellarai_should_rebuild_fleet = yes
		}
		# spend more on ship upgrades if at capacity
		modifier = {
			factor = 2.5
			used_naval_capacity_percent >= 1
		}
		# Convert a stable research lead into current military power.
		modifier = {
			factor = 1.5
			stellarai_research_ready = yes
			has_monthly_income = {
				resource = alloys
				value > 25
			}
			resource_stockpile_compare = {
				resource = alloys
				value > 1200
			}
		}
		# Biological Ships use less alloys
		modifier = {
			factor = 0.33
			country_uses_bio_ships = yes
		}
	}
}

alloys_expenditure_planets = {
	resource = alloys
	type = expenditure
	category = planets
	potential = {
		always = yes
	}
	weight = {
		weight = 0.1
		# During wars, avoid letting planet alloy spending compete too much with shipbuilding
		# while the navy is still below desired strength.
		modifier = {
			factor = 0.4
			is_at_war = yes
			used_naval_capacity_percent < 1.1
		}
		modifier = {
			factor = 5
			is_nomadic = yes
		}
	}
	desired_min = {
		modifier = {
			add = 250
			any_owned_planet = {
				is_planet_class = pc_habitat
				free_district_slots > 0
			}
		}
		modifier = {
			add = 500
			any_owned_planet = {
				is_planet_class = pc_city
				free_district_slots > 0
			}
		}
		modifier = {
			add = 250
			is_void_dweller_empire = yes
		}
		modifier = {
			add = @ai_nomad_alloys_planet_desired_min
			is_nomadic = yes
		}
	}
	desired_max = {
		base = 5000
	}
}

alloys_expenditure_starbases = {
	resource = alloys
	type = expenditure
	category = starbases
	potential = {
		NOT = {
			is_country_type = fallen_empire
		}
	}
	weight = {
		weight = 0.5
		# spend more on starbase upgrades if at fleet cap
		modifier = {
			factor = 2
			used_naval_capacity_percent >= 1
		}
		# Reduce spending on claiming new systems if we have unclaimed planets
		modifier = {
			is_machine_empire = yes
			has_country_resource = {
				type = alloys
				amount < 450
			}
			factor = 0.5
			ai_colonize_plans > 0
		}
		# Reduce spending on claiming new systems if we have unclaimed planets
		modifier = {
			is_machine_empire = no
			has_country_resource = {
				type = alloys
				amount < 250
			}
			factor = 0.75
			ai_colonize_plans > 0
		}
		modifier = {
			factor = 1.2
			is_machine_empire = yes
		}
		modifier = {
			factor = 0.45
			is_at_war = no
			stellarai_machine_energy_emergency = yes
		}
		modifier = {
			factor = 1.35
			has_valid_civic = civic_machine_terminator
		}
		modifier = {
			factor = 1.2
			stellarai_machine_expand_ok = yes
		}
		modifier = {
			factor = 1.75
			used_naval_capacity_percent > 0.75
			any_owned_starbase = {
				can_be_upgraded = yes
			}
		}
		modifier = {
			factor = 1.25
			highest_threat > 20
			OR = {
				any_owned_starbase = {
					can_be_upgraded = yes
				}
				any_owned_nonprimary_starbase = {
					can_be_upgraded = yes
				}
			}
		}
		modifier = {
			factor = 1.5
			OR = {
				any_owned_starbase = {
					NOT = {
						has_starbase_size = starbase_outpost
					}
					can_be_upgraded = yes
				}
				any_owned_nonprimary_starbase = {
					can_be_upgraded = yes
				}
			}
		}
		# Biological Starbases use less alloys
		modifier = {
			factor = 0.5
			country_uses_bio_ships = yes
		}
		modifier = {
			factor = 2
			is_nomadic = yes
		}
	}
	desired_max = {
		base = 500
		modifier = {
			# Allow saving enough for a Starhold (1000 alloys)
			add = 500
			country_uses_bio_ships = no
			has_technology = tech_starbase_3
			any_owned_starbase = {
				has_starbase_size = starbase_starport
				can_be_upgraded = yes
			}
		}
		# Bioship Starhold costs 500 alloys - base 500 is sufficient, no add needed
		modifier = {
			# Allow saving enough for a Starfortress (2500 alloys)
			add = 2000
			country_uses_bio_ships = no
			has_technology = tech_starbase_4
			any_owned_starbase = {
				has_starbase_size = starbase_starhold
				can_be_upgraded = yes
			}
		}
		modifier = {
			# Allow saving enough for a Bioship Starfortress (1250 alloys)
			add = 750
			country_uses_bio_ships = yes
			has_technology = tech_starbase_4
			any_owned_starbase = {
				has_starbase_size = starbase_starhold
				can_be_upgraded = yes
			}
		}
		modifier = {
			# Allow saving enough for a Citadel (5000 alloys)
			add = 4500
			country_uses_bio_ships = no
			has_technology = tech_starbase_5
			any_owned_starbase = {
				has_starbase_size = starbase_starfortress
				can_be_upgraded = yes
			}
		}
		modifier = {
			# Allow saving enough for a Bioship Citadel (2500 alloys)
			add = 2000
			country_uses_bio_ships = yes
			has_technology = tech_starbase_5
			any_owned_starbase = {
				has_starbase_size = starbase_starfortress
				can_be_upgraded = yes
			}
		}
		modifier = {
			# Allow saving enough for an Arkship tier 2 upgrade (5000 alloys)
			add = 5000
			is_nomadic = yes
			has_technology = tech_arkship_tier_2
			any_owned_starbase = {
				is_arkship_starbase = yes
				can_be_upgraded = yes
			}
		}
		modifier = {
			# Allow saving enough for an Arkship tier 3 upgrade (5000 alloys)
			add = 5000
			is_nomadic = yes
			has_technology = tech_arkship_tier_3
			any_owned_starbase = {
				is_arkship_starbase = yes
				can_be_upgraded = yes
			}
		}
		modifier = {
			add = 5000
			has_ascension_perk = ap_cosmogenesis
		}
	}
}

	resource = alloys
	type = expenditure
	category = starbases
	potential = {
		has_ai_expansion_plan = yes
		highest_threat < 50
		NOT = {
			is_country_type = fallen_empire
			is_country_type = awakened_fallen_empire
			ai_colonize_plans > 0
		}
		has_resource = {
			type = influence
			amount > 75
		}
	}
	weight = {
		weight = 0.35
		# In wartime, expansion starbase spending should not compete strongly with shipbuilding
		# unless the AI is already filling its navy.
		modifier = {
			factor = 0.65
			is_at_war = yes
			used_naval_capacity_percent < 0.95
		}
		modifier = {
			factor = 0.55
			AND = {
				ai_colonize_plans > 0
				is_machine_empire = no
				NOT = {
					stellarai_military_emergency = yes
				}
			}
		}
		modifier = {
			factor = 0.8
			AND = {
				ai_colonize_plans > 0

					has_valid_civic = civic_machine_terminator

			factor = 0.5

		}

	desired_min = {
		base = 150
	}
}

			}
		}
	}
	desired_max = {
		base = 20000		# current max alloy cost for a stage
	}
}
